    ```go
    package aws

    import (
        "context"
        "os"
        "time"

        "github.com/aws/aws-sdk-go-v2/aws"
        "github.com/aws/aws-sdk-go-v2/config"

    )

    // SessionRecord is written to the queue-sessions DynamoDB table per D-20.
    type SessionRecord struct {
        TicketID  string
        EventID   string
        JTI       string
        AdmittedAt time.Time
    }

    type DynamoWriter struct {
        client    *dynamodb.Client
        tableName string
    }

    // NewDynamoWriter returns nil when AWS_REGION is unset (local dev skip guard per D-20).
    func NewDynamoWriter(tableName string) (*DynamoWriter, error) {
        if os.Getenv("AWS_REGION") == "" {
            return nil, nil // ponytail: nil receiver = no-op; local dev never sets AWS_REGION
        }
        cfg, err := config.LoadDefaultConfig(context.Background())
        if err != nil {
            return nil, err
        }
        return &DynamoWriter{client: dynamodb.NewFromConfig(cfg), tableName: tableName}, nil
    }

    // Write writes a session record; no-ops when receiver is nil (D-20 skip guard).
    func (dw *DynamoWriter) Write(ctx context.Context, r SessionRecord) error {
        if dw == nil {
            return nil
        }
        _, err := dw.client.PutItem(ctx, &dynamodb.PutItemInput{
            TableName: aws.String(dw.tableName),
            Item: map[string]types.AttributeValue{
                "ticketId":   &types.AttributeValueMemberS{Value: r.TicketID},
                "eventId":    &types.AttributeValueMemberS{Value: r.EventID},
                "jti":        &types.AttributeValueMemberS{Value: r.JTI},
                "admittedAt": &types.AttributeValueMemberS{Value: r.AdmittedAt.UTC().Format(time.RFC3339)},
            },
        })
        return err
    }
    ```

    Create `internal/aws/sqs.go`:

    ```go
    package aws

    import (
        "context"
        "encoding/json"
        "os"

        "github.com/aws/aws-sdk-go-v2/aws"
        "github.com/aws/aws-sdk-go-v2/config"
        "github.com/aws/aws-sdk-go-v2/service/sqs"
    )

    // AdmissionEvent is published to the SQS FIFO queue per D-21.
    type AdmissionEvent struct {
        TicketID string `json:"ticketId"`
        EventID  string `json:"eventId"`
        JTI      string `json:"jti"`
    }

    type SQSEmitter struct {
        client   *sqs.Client
        queueURL string
    }

    // NewSQSEmitter returns nil when AWS_REGION is unset (local dev skip guard per D-21).
    func NewSQSEmitter(queueURL string) (*SQSEmitter, error) {
        if os.Getenv("AWS_REGION") == "" {
            return nil, nil // ponytail: nil receiver = no-op
        }
        cfg, err := config.LoadDefaultConfig(context.Background())
        if err != nil {
            return nil, err
        }
        return &SQSEmitter{client: sqs.NewFromConfig(cfg), queueURL: queueURL}, nil
    }

    // Emit publishes one admission event; no-ops when receiver is nil (D-21 skip guard).
    func (se *SQSEmitter) Emit(ctx context.Context, e AdmissionEvent) error {
        if se == nil {
            return nil
        }
        body, err := json.Marshal(e)
        if err != nil {
            return err
        }
        _, err = se.client.SendMessage(ctx, &sqs.SendMessageInput{
            QueueUrl:       aws.String(se.queueURL),
            MessageBody:    aws.String(string(body)),
            MessageGroupId: aws.String(e.EventID), // FIFO group per eventId
        })
        return err
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub Actions → AWS | Credentials in GitHub Secrets; least-privilege IAM user for CI |
| scheduler → DynamoDB | Write-only (PutItem); no read/scan/delete from application code |
| scheduler → SQS | SendMessage only; cannot read or delete messages |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-03-09 | Information Disclosure | GitHub Actions secrets | high | mitigate | AWS credentials stored as GitHub encrypted secrets, never in workflow YAML; IAM user for CI has minimal permissions (ECR push + ECS update-service only) |
| T-03-10 | Tampering | DynamoDB audit records | low | accept | Scheduler only has PutItem; records are immutable audit trail; deletion requires separate IAM permission not granted |
| T-03-11 | Repudiation | SQS admission events | low | accept | FIFO + content-based dedup prevents duplicate messages; MessageGroupId=eventId preserves ordering; audit trail in DynamoDB is the authoritative record |
| T-03-12 | Denial of Service | SQS message flood | low | accept | Admission rate is scheduler-controlled (1 tick/sec, configurable); SQS SendMessage rate bounded by admission rate |
| T-03-SC | Tampering | npm/pip/cargo installs | low | accept | Go module hashes in go.sum verify all dependencies; no new packages without go.sum update |
</threat_model>
