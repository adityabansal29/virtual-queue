    Function logic — place in `infra/modules/cloudfront/functions/stub_origin_guard.js`. Uses Terraform `templatefile()` to inject `kvs_id` and `queue_api_alb_dns` at plan time (Terraform template variable syntax `${varname}`):

    ```javascript
    // CloudFront Function — stub origin edge JWT enforcement (D-08, D-09, D-10)
    // Replicates the JWT verify portion of pkg/middleware/queue_guard.go at edge.
    // SETNX one-time enforcement stays in Go queue_guard.go (D-12).
    import cf from "cloudfront";

    // Constants injected by Terraform templatefile() — must be declared before handler.
    const KVS_ID        = "${kvs_id}";          // CloudFront KVS resource ID
    const QUEUE_JOIN_HOST = "${queue_api_alb_dns}"; // Queue API ALB DNS for 302 redirect target

    async function handler(event) {
      const request = event.request;
      const cookies = request.cookies;

      // Read secrets from CloudFront KVS (D-10). cf.kvs() is synchronous; .get() is async.
      const kvsHandle = cf.kvs(KVS_ID);
      let admissionSecret, sessionSecret;
      try {
        admissionSecret = await kvsHandle.get("ADMISSION_SECRET");
        sessionSecret   = await kvsHandle.get("SESSION_SECRET");
      } catch (e) {
        // KVS unavailable — fail open to avoid blocking all traffic in misconfiguration
        return request;
      }

      // Try q_session first (fast path — already admitted this session)
      const qSession = cookies["q_session"]?.value;
      if (qSession && await verifyHMACSHA256JWT(qSession, sessionSecret)) {
        return request;
      }

      // Try q_admission (first admission)
      const qAdmission = cookies["q_admission"]?.value;
      if (qAdmission && await verifyHMACSHA256JWT(qAdmission, admissionSecret)) {
        return request; // SETNX enforcement happens at Go origin (D-12)
      }

      // No valid token — 302 to queue join (D-09)
      const uri    = request.uri;
      const qs     = request.querystring ? "?" + request.querystring : "";
      const target = encodeURIComponent(uri + qs);
      return {
        statusCode: 302,
        statusDescription: "Found",
        headers: {
          location: { value: "https://" + QUEUE_JOIN_HOST + "/queue/join?eventId=evt-001&target=" + target }
        }
      };
    }

    // verifyHMACSHA256JWT: verifies signature only — origin Go code checks expiry (D-08).
    async function verifyHMACSHA256JWT(tokenStr, secret) {
      try {
        const parts = tokenStr.split(".");
        if (parts.length !== 3) return false;
        const data = new TextEncoder().encode(parts[0] + "." + parts[1]);
        const sig  = base64urlDecode(parts[2]);
        const key  = await crypto.subtle.importKey(
          "raw", new TextEncoder().encode(secret),
          { name: "HMAC", hash: "SHA-256" }, false, ["verify"]
        );
        return await crypto.subtle.verify("HMAC", key, sig, data);
      } catch (_) {
        return false;
      }
    }

    function base64urlDecode(str) {
      const b64 = str.replace(/-/g, "+").replace(/_/g, "/")
                     .padEnd(str.length + (4 - str.length % 4) % 4, "=");
      const bin = atob(b64);
      return Uint8Array.from(bin, c => c.charCodeAt(0));
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| public internet → stub origin | CloudFront is the only entry; ALB SG blocks direct access (D-13) |
| CloudFront Function → KVS | Secrets read at request time; rotatable without function redeploy (D-10) |
| CloudFront Function → request | Edge verifies JWT signature only; origin Go code does SETNX single-use enforcement (D-12) |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-03-17 | Spoofing | CloudFront Function JWT verify | high | mitigate | D-08: HMAC-SHA256 verify via crypto.subtle — cannot forge without ADMISSION_SECRET or SESSION_SECRET; KVS is the only storage of these secrets at edge (D-10) |
| T-03-18 | Tampering | CloudFront KVS secret values | high | mitigate | D-10: KVS populated via CLI from SSM SecureString source; secret values never appear in Terraform state or workflow logs; KVS access requires AWS IAM; update requires KVS ETag (optimistic lock) |
| T-03-19 | Bypass | CloudFront Function fail-open on KVS error | medium | accept | Function fails open (passes request) when KVS is unavailable — avoids total outage. Go queue_guard.go provides the fallback enforcement layer (D-12). Acceptable for a learning project; production would fail-closed |
| T-03-20 | Elevation of Privilege | Direct ALB access (bypassing CF) | critical | mitigate | D-13: ALB SG restricted to CloudFront managed prefix list; direct internet access blocked at network layer; defence-in-depth with Go queue_guard.go still enforcing SETNX |
| T-03-21 | Information Disclosure | QUEUE_JOIN_HOST in CF Function code | low | accept | ALB DNS is not sensitive; it is public (resolvable); embedding it in CF Function code is acceptable. Route 53 / custom domain would be the proper solution (D-14 deferred) |
| T-03-SC | Tampering | No npm/pip/cargo installs | low | accept | CloudFront Function uses built-in cloudfront-js-2.0 runtime; no package installs |
